def _is_safe_path(path: Path, project_root: Path) -> tuple[bool, str]:
    """
    Check if a path is safe to read.
    
    Returns:
        (is_safe, error_message)
    """
    try:
        # Resolve to absolute path
        abs_path = path.resolve()
        abs_root = project_root.resolve()
        
        # Check if path is within project root
        try:
            abs_path.relative_to(abs_root)
        except ValueError:
            return False, f"Access denied: path outside project root ({abs_root})"
        
        # Check for blocked patterns
        path_str = str(abs_path).lower()
        for pattern in BLOCKED_PATTERNS:
            if pattern in path_str:
                return False, f"Access denied: blocked file pattern '{pattern}'"
        
        return True, ""
    except Exception as e:
        return False, f"Path validation error: {str(e)}"


def _read_file_safe(path: Path, max_bytes: int, project_root: Path) -> ReadFileResult:
    """
    Safely read a file with security checks.
    
    Args:
        path: Path to read
        max_bytes: Maximum bytes to read
        project_root: Project root directory
        
    Returns:
        ReadFileResult with content or error
    """
    # Security check
    is_safe, error_msg = _is_safe_path(path, project_root)
    if not is_safe:
        return ReadFileResult(
            success=False,
            path=str(path),
            error=error_msg
        )
    
    # Check if file exists
    if not path.exists():
        return ReadFileResult(
            success=False,
            path=str(path),
            error=f"File not found: {path}"
        )
    
    # Check if it's a file
    if not path.is_file():
        return ReadFileResult(
            success=False,
            path=str(path),
            error=f"Not a file: {path}"
        )
    
    try:
        # Get file size
        size_bytes = path.stat().st_size
        
        # Read file content
        with open(path, 'r', encoding='utf-8', errors='replace') as f:
            content = f.read(max_bytes)
        
        # Check if truncated
        truncated = size_bytes > max_bytes
        
        return ReadFileResult(
            success=True,
            path=str(path),
            content=content,
            truncated=truncated,
            size_bytes=size_bytes
        )
    except UnicodeDecodeError:
        return ReadFileResult(
            success=False,
            path=str(path),
            error="File is not a text file (binary content detected)"
        )
    except PermissionError:
        return ReadFileResult(
            success=False,
            path=str(path),
            error="Permission denied"
        )
    except Exception as e:
        return ReadFileResult(
            success=False,
            path=str(path),

def is_path_safe(file_path: Path, project_root: Path) -> tuple[bool, Optional[str]]:
    """
    Check if a file path is safe to read.
    
    Security checks:
    - Path must be within project root (no directory traversal)
    - Path must not match blocked file patterns (secrets, credentials, etc.)
    
    Args:
        file_path: Path to check
        project_root: Project root directory
        
    Returns:
        (is_safe, error_message) - True if safe, False with reason if blocked
    """
    try:
        # Resolve to absolute path
        abs_path = file_path.resolve()
        abs_root = project_root.resolve()
        
        # Check if path is within project root (prevent directory traversal)
        try:
            abs_path.relative_to(abs_root)
        except ValueError:
            return False, f"Access denied: path outside project root ({abs_root})"
        
        # Check for blocked file patterns
        path_str = str(abs_path).lower()
        path_name = abs_path.name.lower()
        
        for pattern in BLOCKED_FILE_PATTERNS:
            # Check both full path and filename
            if pattern in path_str or pattern in path_name:
                return False, f"Access denied: blocked file pattern '{pattern}'"
        
        return True, None
        
    except Exception as e:
        return False, f"Path validation error: {str(e)}"

### Description

Safely read local files (reports, logs, chaos results) after chaos runs. Includes security checks to block reading outside project root and sensitive files. Returns file content with truncation metadata.

### Use Cases

- Reading chaos test reports and results
- Analyzing application logs after chaos runs
- Inspecting crash dumps and error logs
- Reviewing generated test files
- Post-chaos analysis and debugging

### Security Features

This tool includes comprehensive security checks:

**Blocked Paths:**
- Paths outside project root (prevents directory traversal attacks)
- Sensitive files: `.env`, `credentials`, `id_rsa`, `tokens`, `secrets`, `.pem`, `.key`, `password`, etc.

- Maximum file size: 1MB (1,000,000 bytes)
